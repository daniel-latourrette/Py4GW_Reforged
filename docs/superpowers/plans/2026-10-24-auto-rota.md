# Auto Rota Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Um widget Py4GW em que cada toque numa tecla lança a melhor jogada de Mesmer: primeiro um interrupt escolhido por prioridades, senão a próxima skill da rotação do HeroAI.

**Architecture:** O núcleo de decisão (`model.py`, `priorities.py`, `decide.py`) é Python puro, sem importar o Py4GWCoreLib, por isso é testado fora do jogo com `unittest`. Só o `game_adapter.py` fala com o jogo e reutiliza o `CacheData`/`CombatClass` do HeroAI e o `is_interrupt_feasible`. O widget `Auto Rota.py` desenha a janela, regista a tecla no `HOTKEY_MANAGER` e liga tudo.

**Tech Stack:** Python 3.13 32 bits dentro do jogo (testes offline com `python3.12`), Py4GWCoreLib, HeroAI, PyImGui, `unittest` e Pyright (`npx -y pyright@1.1.390`).

**Spec:** `docs/superpowers/specs/2026-10-24-auto-rota-design.md`

> **Estado (2026-10-24):** plano executado. As revisões por tarefa mudaram o código em relação aos blocos abaixo: `CombatClass` privado em vez do `combat_handler` partilhado, sem `refresh()`, ciclo ordenado em `choose_interrupt`, filtros da tecla, pré-visualização sem reservas de alvo e mais 3 testes. A spec atualizada e o código são a referência; este plano fica como registo histórico.

**Desvios da spec, decididos ao escrever o plano:**
- Os testes usam `unittest`, não pytest. É a convenção do repo (`Examples and tests/tests/`) e o `docs/py4gw-ai` proíbe introduzir outro framework de testes. Ficheiro: `Examples and tests/tests/auto_rota/test_auto_rota_decide.py`.
- O `priorities.py` guarda os **nomes** dos atributos de cura (strings) e não o enum `Attribute`. O núcleo fica sem dependências do jogo e o adapter converte os nomes com `Attribute[name]`.
- Uma skill conta como spell se o tipo for Spell, Hex, Enchantment, Well, Ward, WeaponSpell ou ItemSpell. O `GLOBAL_CACHE.Skill.Flags.IsSpell` só reconhece o tipo "Spell" e falharia hexes e encantamentos.
- A instância privada do `CombatClass` liga `is_targeting_enabled = True`, para a rotação escolher alvos adequados (aliado para curas, inimigo para ofensivas) em vez de usar sempre o alvo atual.

**Regra do repo sobre commits:** o `AGENTS.md` proíbe commits sem pedido explícito do utilizador. Os passos "Commit" só se executam com autorização dada no chat, uma vez para o plano inteiro ou tarefa a tarefa.

This module must not import Py4GWCoreLib: the decision core runs offline in
unit tests, outside the injected Guild Wars runtime.

from __future__ import annotations

from dataclasses import dataclass
from enum import IntEnum


class CastKind(IntEnum):
    SPELL = 0
    CHANT = 1
    ATTACK = 2
    OTHER = 3


class InterruptReach(IntEnum):
    ANY_CAST = 0
    SPELL = 1
    SPELL_OR_CHANT = 2
    ATTACK = 3


@dataclass(frozen=True)
class EnemyCast:
    agent_id: int
    skill_id: int
    kind: CastKind
    is_resurrection: bool
    is_healing: bool
    is_elite: bool
    activation_ms: int
    remaining_ms: int


@dataclass(frozen=True)
class InterruptSlot:
    slot: int  # index in CombatClass.skills (priority order)
    bar_slot: int  # 1-based skillbar slot passed to SkillBar.UseSkill
    skill_id: int
    reach: InterruptReach


@dataclass(frozen=True)
class Decision:
    bar_slot: int
    skill_id: int
    target_id: int
    label: str
    enemy_skill_id: int = 0

def can_hit(reach: InterruptReach, kind: CastKind) -> bool:
    if reach == InterruptReach.ANY_CAST:
        return True
    if reach == InterruptReach.SPELL:
        return kind == CastKind.SPELL
    if reach == InterruptReach.SPELL_OR_CHANT:
        return kind in (CastKind.SPELL, CastKind.CHANT)
    if reach == InterruptReach.ATTACK:
        return kind == CastKind.ATTACK
    raise ValueError(f"Unhandled InterruptReach: {reach!r}")

FeasibilityCheck = Callable[[InterruptSlot, EnemyCast], bool]


def choose_interrupt(
    casts: Sequence[EnemyCast],
    slots: Sequence[InterruptSlot],
    feasible: FeasibilityCheck,
) -> Optional[Decision]:
    """Pick the best (interrupt, cast) pair.

    Order: lowest tier value first, then the cast that finishes first. For the
    chosen cast, the first compatible and feasible slot in bar order is used.
    """
    best: Optional[Decision] = None
    best_key: Optional[tuple[int, int]] = None
    for cast in casts:
        tier = classify(cast)
        key = (int(tier), cast.remaining_ms)
        if best_key is not None and key >= best_key:
            continue
        for slot in slots:
            if not can_hit(slot.reach, cast.kind) or not feasible(slot, cast):
                continue
            best = Decision(
                bar_slot=slot.bar_slot,
                skill_id=slot.skill_id,
                target_id=cast.agent_id,
                label=f"Interrupt [{TIER_LABELS[tier]}]",
                enemy_skill_id=cast.skill_id,
            )
            best_key = key
            break
    return best
```

    Skilltarget.EnemyCastingSpellOrChant.value: InterruptReach.SPELL_OR_CHANT,
    Skilltarget.EnemyAttacking.value: InterruptReach.ATTACK,
}

# Flags.IsSpell only matches the plain "Spell" type; hexes, enchantments,
# wells, wards and weapon/item spells are spells for interrupt purposes too.
_SPELL_TYPES: frozenset[int] = frozenset(
    {
        SkillType.Spell.value,
        SkillType.Hex.value,
        SkillType.Enchantment.value,
        SkillType.Well.value,
        SkillType.Ward.value,
        SkillType.WeaponSpell.value,
        SkillType.ItemSpell.value,
    }
)

_HEALING_ATTRIBUTE_IDS: frozenset[int] = frozenset(Attribute[name].value for name in HEALING_ATTRIBUTE_NAMES)


def _cast_kind(skill_id: int) -> CastKind:
    skill_type, _ = GLOBAL_CACHE.Skill.GetType(skill_id)
    if skill_type == SkillType.Attack.value:
        return CastKind.ATTACK
    if skill_type == SkillType.Chant.value:
        return CastKind.CHANT
    if skill_type in _SPELL_TYPES:
        return CastKind.SPELL
    return CastKind.OTHER


def _skill_name(skill_id: int) -> str:
    name = GLOBAL_CACHE.Skill.GetName(skill_id)
    return name.replace("_", " ") if name else f"skill {skill_id}"


class GameAdapter:
    def __init__(self, cached_data: CacheData) -> None:
        self._cached_data = cached_data
        self._last_execute_s: float = 0.0

    def refresh(self) -> None:
        """Per-frame: rebuild the private CombatClass view of the skillbar."""
        self._cached_data.UpdateCombat()
        combat = self._cached_data.combat_handler
        combat.is_skill_enabled = [True] * MAX_SKILLS
        combat.is_targeting_enabled = True

    def heroai_conflict(self) -> bool:
        from Py4GW_widget_manager import get_widget_handler

        if not get_widget_handler().is_widget_enabled("HeroAI"):
            return False
        options = self._cached_data.account_options
        return bool(options is not None and options.Combat)

    def blocker(self) -> Optional[str]:
        """Return why a key press must be refused right now, or None."""
        if self.heroai_conflict():
            return REASON_HEROAI
        if not Routines.Checks.Map.IsExplorable():
            return REASON_NOT_EXPLORABLE
        player_id = Player.GetAgentID()
        if not Agent.IsAlive(player_id):
            return REASON_DEAD
        if Agent.IsCasting(player_id) or (GLOBAL_CACHE.SkillBar.GetCasting() or 0) != 0:
            return REASON_BUSY
        if (time.monotonic() - self._last_execute_s) * 1000.0 < _DEBOUNCE_MS:
            return REASON_BUSY
        return None

    def enemy_casts(self) -> list[EnemyCast]:
        player_xy = Player.GetXY()
        casts: list[EnemyCast] = []
        for agent_id in AgentArray.GetEnemyArray():
            try:
                if not Agent.IsAlive(agent_id) or not Agent.IsCasting(agent_id):
                    continue
                if Utils.Distance(player_xy, Agent.GetXY(agent_id)) > Range.Longbow.value:
                    continue
                skill_id = Agent.GetCastingSkillID(agent_id)
                if skill_id == 0:
                    continue
                casts.append(self._describe_cast(agent_id, skill_id))
            except Exception as exc:
                ConsoleLog(LOG_SENDER, f"enemy_casts: ignoring agent {agent_id}: {exc}", Console.MessageType.Warning)
        return casts

    def _describe_cast(self, agent_id: int, skill_id: int) -> EnemyCast:
        activation_ms = int((GLOBAL_CACHE.Skill.Data.GetActivation(skill_id) or 0.0) * 1000)
        elapsed_ms = cast_observer.elapsed_ms(agent_id, skill_id) or 0
        nature = custom_skill_data_handler.get_skill(skill_id).Nature
        attribute_id = int(GLOBAL_CACHE.Skill.Attribute.GetAttribute(skill_id))
        return EnemyCast(
            agent_id=agent_id,
            skill_id=skill_id,
            kind=_cast_kind(skill_id),
            is_resurrection=nature == SkillNature.Resurrection.value,
            is_healing=nature == SkillNature.Healing.value or attribute_id in _HEALING_ATTRIBUTE_IDS,
            is_elite=bool(GLOBAL_CACHE.Skill.Flags.IsElite(skill_id)),
            activation_ms=activation_ms,
            remaining_ms=max(0, activation_ms - elapsed_ms),
        )

    def interrupt_slots(self) -> list[InterruptSlot]:
        combat = self._cached_data.combat_handler
        slots: list[InterruptSlot] = []
        for index, skill in enumerate(combat.GetSkills()):
            data = skill.custom_skill_data
            if data.Nature != SkillNature.Interrupt.value:
                continue
            reach = _REACH_BY_ALLEGIANCE.get(int(data.TargetAllegiance))
            if reach is None:
                continue
            if not combat.IsSkillReady(index):
                continue
            if not self._has_resources(skill.skill_id, int(skill.skillbar_data.adrenaline_a)):
                continue
            slots.append(
                InterruptSlot(slot=index, bar_slot=combat.skill_order[index] + 1, skill_id=skill.skill_id, reach=reach)
            )
        return slots

    def _has_resources(self, skill_id: int, adrenaline_a: int) -> bool:
        # Mirrors the adrenaline/energy gates of CombatClass.IsReadyToCast,
        # which cannot be reused here because it also picks its own target.
        combat = self._cached_data.combat_handler
        player_id = Player.GetAgentID()
        adrenaline_required = GLOBAL_CACHE.Skill.Data.GetAdrenaline(skill_id)
        if adrenaline_required > 0 and adrenaline_a < adrenaline_required:
            return False
        energy_fraction = combat.GetEnergyValues(player_id)
        if energy_fraction < 0.0:
            return False
        current_energy = energy_fraction * Agent.GetMaxEnergy(player_id)
        energy_cost = Routines.Checks.Skills.GetEnergyCostWithEffects(skill_id, player_id)
        if combat.expertise_exists:
            energy_cost = Routines.Checks.Skills.apply_expertise_reduction(energy_cost, combat.expertise_level, skill_id)
        return current_energy >= energy_cost

    def is_feasible(self, slot: InterruptSlot, cast: EnemyCast) -> bool:
        combat = self._cached_data.combat_handler
        return is_interrupt_feasible(
            target_agent_id=cast.agent_id,
            our_skill_id=slot.skill_id,
            fast_casting_level=combat.fast_casting_level,
            ping_ms=int(combat.ping_handler.GetCurrentPing()),
        )

    def rotation_decision(self) -> Optional[Decision]:
        combat = self._cached_data.combat_handler
        for index, skill in enumerate(combat.GetSkills()):
            if skill.custom_skill_data.Nature == SkillNature.Interrupt.value:
                continue
            if not combat.IsSkillReady(index):
                continue
            ready, target_id = combat.IsReadyToCast(index)
            if not ready or target_id == 0 or not Agent.IsLiving(target_id):
                continue
            return Decision(
                bar_slot=combat.skill_order[index] + 1, skill_id=skill.skill_id, target_id=target_id, label="Rotacao"
            )
        return None

    def next_decision(self) -> Optional[Decision]:
        interrupt = choose_interrupt(self.enemy_casts(), self.interrupt_slots(), self.is_feasible)
        if interrupt is not None:
            return interrupt
        return self.rotation_decision()

    def execute(self, decision: Decision) -> None:
        GLOBAL_CACHE.SkillBar.UseSkill(decision.bar_slot, decision.target_id, aftercast_delay=_AFTERCAST_MS)
        self._last_execute_s = time.monotonic()

    def describe(self, decision: Decision) -> str:
        target_name = Agent.GetNameByID(decision.target_id) or f"agente {decision.target_id}"
        text = f"{_skill_name(decision.skill_id)} -> {target_name}"
        if decision.enemy_skill_id:
            text += f" ({_skill_name(decision.enemy_skill_id)})"
        return f"{text} [{decision.label}]"
```

"""Auto Rota: one key press, one best play (Mesmer-focused interrupts first)."""

from __future__ import annotations

import time
import traceback
from collections import deque

import PyImGui

from Py4GWCoreLib import Routines
from Py4GWCoreLib import Timer
from Py4GWCoreLib.enums_src.IO_enums import Key
from Py4GWCoreLib.HeroAI.cache_data import CacheData
from Py4GWCoreLib.HotkeyManager import HOTKEY_MANAGER
from Py4GWCoreLib.py4gwcorelib_src.Console import Console
from Py4GWCoreLib.py4gwcorelib_src.Console import ConsoleLog
from Py4GWCoreLib.py4gwcorelib_src.Settings import Settings
from Widgets.Automation.Helpers.AutoRota.auto_rota_src.game_adapter import REASON_BUSY
from Widgets.Automation.Helpers.AutoRota.auto_rota_src.game_adapter import GameAdapter

MODULE_NAME = "Auto Rota"

_HOTKEY_ID = "auto_rota.play"
_SETTINGS_SECTION = "AutoRota"
_SETTINGS_HOTKEY = "hotkey"
_PREVIEW_INTERVAL_MS = 100
_HISTORY_SIZE = 5
_WARNING_COLOR = (1.0, 0.35, 0.35, 1.0)
_KEY_CHOICES: list[tuple[str, Key]] = [
    ("` (acento grave)", Key.VK_OEM_3),
    ("F1", Key.VK_F1),
    ("F2", Key.VK_F2),
    ("F3", Key.VK_F3),
    ("F4", Key.VK_F4),
    ("F5", Key.VK_F5),
]

_cached_data = CacheData()
_adapter = GameAdapter(_cached_data)
_settings = Settings("Widgets/Config/auto_rota.ini", "global")
_preview_timer = Timer()
_preview_timer.Start()
_preview_text = "-"
_history: deque[str] = deque(maxlen=_HISTORY_SIZE)
_hotkey_registered = False
_key_index = 0


def _log_error(stage: str, exc: Exception) -> None:
    ConsoleLog(MODULE_NAME, f"{stage}: {exc}\n{traceback.format_exc()}", Console.MessageType.Error)


def _push_history(text: str) -> None:
    _history.appendleft(f"{time.strftime('%H:%M:%S')}  {text}")


def _load_key_index() -> int:
    saved = _settings.get_str(_SETTINGS_SECTION, _SETTINGS_HOTKEY, Key.VK_OEM_3.name)
    for index, (_, key) in enumerate(_KEY_CHOICES):
        if key.name == saved:
            return index
    return 0


def _on_hotkey() -> None:
    try:
        reason = _adapter.blocker()
        if reason is not None:
            _push_history(f"recusado: {reason}")
            return
        decision = _adapter.next_decision()
        if decision is None:
            _push_history("nada a fazer")
            return
        _adapter.execute(decision)
        _push_history(_adapter.describe(decision))
    except Exception as exc:
        _push_history(f"erro: {exc}")
        _log_error("hotkey", exc)


def _register_hotkey() -> None:
    global _hotkey_registered
    # Re-registering the same identifier replaces the previous key binding.
    HOTKEY_MANAGER.register_hotkey(_KEY_CHOICES[_key_index][1], _HOTKEY_ID, _on_hotkey, name="Auto Rota: jogada")
    _hotkey_registered = True


def _update_preview() -> None:
    global _preview_text
    reason = _adapter.blocker()
    if reason is not None and reason != REASON_BUSY:
        _preview_text = f"em pausa ({reason})"
        return
    decision = _adapter.next_decision()
    _preview_text = _adapter.describe(decision) if decision is not None else "nada a fazer"


def _draw_window() -> None:
    global _key_index
    if PyImGui.begin(MODULE_NAME, int(PyImGui.WindowFlags.AlwaysAutoResize)):
        if _adapter.heroai_conflict():
            PyImGui.text_colored("HeroAI com combate ligado: Auto Rota em pausa", _WARNING_COLOR)
        PyImGui.text(f"Proxima jogada: {_preview_text}")
        PyImGui.separator()
        labels = [label for label, _ in _KEY_CHOICES]
        new_index = PyImGui.combo("Tecla", _key_index, labels)
        if new_index != _key_index:
            _key_index = new_index
            _settings.set_str(_SETTINGS_SECTION, _SETTINGS_HOTKEY, _KEY_CHOICES[new_index][1].name)
            _register_hotkey()
        PyImGui.separator()
        PyImGui.text("Historico:")
        for line in _history:
            PyImGui.text(line)
    PyImGui.end()


def tooltip() -> None:
    PyImGui.begin_tooltip()
    PyImGui.text("Auto Rota")
    PyImGui.separator()
    PyImGui.text("Cada toque na tecla faz uma jogada:")
    PyImGui.bullet_text("1. Interrupt por prioridade (ressurreicao, cura, elite, cast longo)")
    PyImGui.bullet_text("2. Senao, a proxima skill da rotacao do HeroAI")
    PyImGui.text("Nao funciona com o combate do HeroAI ligado.")
    PyImGui.end_tooltip()


def on_disable() -> None:
    global _hotkey_registered
    HOTKEY_MANAGER.unregister_hotkey(_HOTKEY_ID)
    _hotkey_registered = False


def main() -> None:
    global _key_index
    try:
        if not Routines.Checks.Map.MapValid():
            return
        _cached_data.Update()
        if not (Routines.Checks.Map.IsMapReady() and Routines.Checks.Party.IsPartyLoaded()):
            return
        if not _hotkey_registered:
            _key_index = _load_key_index()
            _register_hotkey()
        _adapter.refresh()
        if _preview_timer.HasElapsed(_PREVIEW_INTERVAL_MS):
            _preview_timer.Reset()
            _update_preview()
        _draw_window()
    except Exception as exc:
        _log_error("main", exc)


if __name__ == "__main__":
    main()
